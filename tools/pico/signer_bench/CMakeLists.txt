# cmake -S . -B build -G Ninja -DPICO_ROOT=<bitwasm-signer-pico> -DPICO_SDK_PATH=<it>/third_party/pico-sdk \
#   -DPICO_TOOLCHAIN_PATH=<it>/third_party/riscv-toolchain -DPOOL_KB=320, with signer_wasm.h from xxd -i -n signer_wasm
cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-riscv)
set(PICO_BOARD pico2)
set(PICO ${PICO_ROOT})
include(${PICO}/third_party/pico-sdk/external/pico_sdk_import.cmake)
project(signer_bench C CXX ASM)
pico_sdk_init()
set(WAMR_ROOT_DIR ${PICO}/third_party/wasm-micro-runtime)
set(SHARED_PLATFORM_CONFIG ${PICO}/src/runtime/wamr-platform/shared_platform.cmake)
set(WAMR_BUILD_PLATFORM rp2350)
set(WAMR_BUILD_TARGET RISCV32_ILP32)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
string(REPLACE "-std=gnu99" "-std=gnu11" CMAKE_C_FLAGS "${CMAKE_C_FLAGS}")
add_executable(signer_bench signer_bench.c ${WAMR_RUNTIME_LIB_SOURCE})
target_compile_definitions(signer_bench PRIVATE PICO_BUILD=1 PICO_STACK_SIZE=0x8000 POOL_KB=${POOL_KB})
target_include_directories(signer_bench PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(signer_bench pico_stdlib)
pico_enable_stdio_uart(signer_bench 1)
pico_enable_stdio_usb(signer_bench 0)
target_link_options(signer_bench PRIVATE -Wl,--print-memory-usage)
pico_add_extra_outputs(signer_bench)
